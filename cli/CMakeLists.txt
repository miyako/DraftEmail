cmake_minimum_required(VERSION 3.16)
set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "Minimum supported macOS")
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(draft-email-cli LANGUAGES CXX)

if(APPLE)
  enable_language(OBJCXX)
  set(BACKEND mac.mm)
elseif(WIN32)
  set(BACKEND windows.cpp)
else()
  message(FATAL_ERROR "Only native macOS and Windows x64 are supported")
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(staging-tests staging_test.cpp staging.cpp protocol.cpp)
  target_compile_features(staging-tests PRIVATE cxx_std_17)
  target_compile_definitions(staging-tests PRIVATE DRAFTEMAIL_STAGING_TEST=1)
  if(APPLE)
    target_link_libraries(staging-tests PRIVATE "-framework Security")
  elseif(WIN32)
    if(MSVC)
      target_compile_options(staging-tests PRIVATE /utf-8)
    endif()
    target_link_libraries(staging-tests PRIVATE advapi32 bcrypt)
  endif()
  add_test(NAME private-staging COMMAND staging-tests)
  add_executable(provider-tests provider_test.cpp)
  target_compile_features(provider-tests PRIVATE cxx_std_17)
  add_test(NAME provider-selection COMMAND provider-tests)
endif()
add_executable(draft-email-cli main.cpp protocol.cpp staging.cpp ${BACKEND})
target_compile_features(draft-email-cli PRIVATE cxx_std_17)
set_target_properties(draft-email-cli PROPERTIES CXX_EXTENSIONS OFF)
set_target_properties(draft-email-cli PROPERTIES OBJCXX_STANDARD 17 OBJCXX_STANDARD_REQUIRED YES)
if(MSVC)
  target_compile_options(draft-email-cli PRIVATE /utf-8)
elseif(MINGW)
  target_link_options(draft-email-cli PRIVATE -municode)
endif()
if(APPLE)
  set_source_files_properties(mac.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
  target_link_libraries(draft-email-cli PRIVATE "-framework AppKit" "-framework Security" "-framework SystemConfiguration")
elseif(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Simple MAPI requires an x64 build")
  endif()
  target_link_libraries(draft-email-cli PRIVATE advapi32 bcrypt user32)
endif()
